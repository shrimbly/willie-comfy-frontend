```typescript
/**
 * Pure filter predicate math (Phase 3).
 *
 * Semantics (CONTEXT.md D-12): OR within a multi-value chip; AND across chips.
 * Semantics (CONTEXT.md D-03 / FILTER-09): missing params silently null — the
 * predicate returns false, so the asset drops from the filtered set.
 *
 * Determinism for testability: `nowMs` is injected — no `Date.now()` inside
 * predicates. Callers (moshpitFilterStore) supply Date.now() at call time.
 *
 * Performance: applyFilterChips is O(N × chips) over N assets. Safe at 5k
 * assets × ≤10 chips × O(1) per-chip cost well inside a 16ms frame budget.
 */

import type { NormalizedParams } from './paramNormalize'
import type { CurationRecord } from './thumbRepository.types'
import type {
  ChipValue,
  FilterChip,
  TimePreset,
  TimeRange
} from './filterTypes'

const DAY_MS = 86_400_000
const DEFAULT_CURATION: CurationRecord = Object.freeze({
  favourite: false,
  tags: [],
  folders: [],
  hidden: false
})

export function getDateRangeForPreset(
  preset: TimePreset,
  nowMs: number
): { readonly from: number; readonly to: number } | null {
  switch (preset) {
    case 'all':
      return null
    case 'today':
      return { from: nowMs - DAY_MS, to: nowMs }
    case 'thisWeek':
      return { from: nowMs - 7 * DAY_MS, to: nowMs }
    case 'thisMonth':
      return { from: nowMs - 30 * DAY_MS, to: nowMs }
    case 'custom':
      return null   // caller uses range.from / range.to directly
  }
}

export function matchesTimeRange(
  timestamp: number,
  range: TimeRange,
  nowMs: number
): boolean {
  if (range.preset === 'all') return true
  if (range.preset === 'custom') {
    const fromOk = range.from === null || timestamp >= range.from
    const toOk = range.to === null || timestamp <= range.to
    return fromOk && toOk
  }
  const window = getDateRangeForPreset(range.preset, nowMs)
  if (!window) return true
  return timestamp >= window.from && timestamp <= window.to
}

export function matchesChip(
  params: NormalizedParams,
  curation: CurationRecord,
  chip: FilterChip
): boolean { /* ... */ }

export function applyFilterChips(
  hashToParams: ReadonlyMap<string, NormalizedParams>,
  hashToCuration: ReadonlyMap<string, CurationRecord>,
  chips: readonly FilterChip[],
  showHidden: boolean,
  timeRange: TimeRange,
  nowMs: number
): readonly string[] { /* ... */ }
```

Implementation notes:

**`matchesChip` dispatch table** — the `chip.value.kind` discriminator + the `chip.param` decides extraction:

- `kind: 'categorical'` + `param: 'loras'`:
  - `return chip.value.values.some(name => params.loras.some(l => l.name === name))` (D-04 name match, any weight; OR within chip)
  - If `params.loras.length === 0` → false

- `kind: 'categorical'` + `param: 'tags'`:
  - `return chip.value.values.some(tag => curation.tags.includes(tag))` (OR within chip)

- `kind: 'categorical'` + other categorical params (model, sampler, scheduler):
  - `const val = params[chip.param]` — narrow via type guard
  - Return false if `val === undefined`
  - `return chip.value.values.includes(val)` (OR within chip via .includes on the values array)

- `kind: 'numeric'`:
  - `const val = params[chip.param]` — where chip.param is one of cfg/steps/seed/width/height — ensure type is `number | undefined`
  - Return false if `val === undefined`
  - If `chip.value.exact !== null` → `return val === chip.value.exact`
  - Otherwise: `const minOk = chip.value.min === null || val >= chip.value.min; const maxOk = chip.value.max === null || val <= chip.value.max; return minOk && maxOk`

- `kind: 'text'`:
  - `const val = chip.param === 'positivePrompt' ? params.positivePrompt : params.negativePrompt` (fallback undefined otherwise)
  - Return false if undefined or empty
  - `return val.toLowerCase().includes(chip.value.substring.toLowerCase())`

- `kind: 'resolution'`:
  - Return false if `params.width === undefined || params.height === undefined`
  - `return chip.value.pairs.some(([w, h]) => params.width === w && params.height === h)`

- `kind: 'boolean'` + `param: 'favourite'`:
  - `return curation.favourite === chip.value.value`

Use a switch on `chip.value.kind` with exhaustiveness guard (`const _exhaustive: never = chip.value`).

```typescript
const result: string[] = []
for (const [hash, p] of hashToParams) {
  const cur = hashToCuration.get(hash) ?? DEFAULT_CURATION

  for (const c of chips) {
    if (!matchesChip(p, cur, c)) { pass = false; break }

  if (pass) result.push(hash)
}
return result
```
