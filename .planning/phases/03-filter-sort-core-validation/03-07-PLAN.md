**A. useMoshpitParamValueOptions composable** — derives discovered categorical values for the value editor.

```ts
// src/platform/moshpit/composables/useMoshpitParamValueOptions.ts
import type { ComputedRef } from 'vue'
import { computed } from 'vue'
import { useMoshpitMetadataStore } from '../stores/moshpitMetadataStore'
import { useMoshpitCurationStore } from '../stores/moshpitCurationStore'
import type { ParamKey } from '../services/filterTypes'

export interface ValueOption {
  readonly value: string
  readonly count: number
}

/**
 * Derives the unique set of values currently present in `paramsByHash` for a
 * given categorical ParamKey (model, sampler, scheduler, loras, tags), along
 * with per-value counts for the picker badge (UI-SPEC §Categorical editor).
 *
 * Returns a ComputedRef<ValueOption[]> sorted by descending count then alpha.
 */
export function useMoshpitParamValueOptions(param: ParamKey | null): {
  readonly options: ComputedRef<readonly ValueOption[]>
  readonly resolutionPairs: ComputedRef<readonly (readonly [number, number])[]>
} {
  const metaStore = useMoshpitMetadataStore()
  const curationStore = useMoshpitCurationStore()

  const options = computed<readonly ValueOption[]>(() => {
    if (!param) return []
    const counts = new Map<string, number>()
    for (const [hash, params] of metaStore.paramsByHash) {
      // eslint-disable-next-line @typescript-eslint/switch-exhaustiveness-check
      switch (param) {
        case 'model':
          if (params.model) counts.set(params.model, (counts.get(params.model) ?? 0) + 1)
          break
        case 'sampler':
          if (params.sampler) counts.set(params.sampler, (counts.get(params.sampler) ?? 0) + 1)
          break
        case 'scheduler':
          if (params.scheduler) counts.set(params.scheduler, (counts.get(params.scheduler) ?? 0) + 1)
          break
        case 'loras':
          for (const l of params.loras) counts.set(l.name, (counts.get(l.name) ?? 0) + 1)
          break
        case 'tags': {
          const cur = curationStore.get(hash)
          if (cur) for (const tag of cur.tags) counts.set(tag, (counts.get(tag) ?? 0) + 1)
          break
        }
        default:
          // numeric/text/boolean/resolution/timestamp params don't use this path
          break
      }
    }
    const out: ValueOption[] = Array.from(counts, ([value, count]) => ({ value, count }))
    out.sort((a, b) => b.count - a.count || (a.value < b.value ? -1 : 1))
    return out
  })

  const resolutionPairs = computed<readonly (readonly [number, number])[]>(() => {
    if (param !== 'resolution') return []
    const seen = new Map<string, [number, number]>()
    for (const [, params] of metaStore.paramsByHash) {
      if (params.width !== undefined && params.height !== undefined) {
        const key = `${params.width}x${params.height}`
        if (!seen.has(key)) seen.set(key, [params.width, params.height])
      }
    }
    return [...seen.values()]
  })

  return { options, resolutionPairs }
}
```

- With 2 assets sampler='euler', 1 asset sampler='dpmpp' → options: euler count 2, dpmpp count 1, euler first
- `param: 'loras'` → one option per unique LoRA name across all assets' loras arrays
- `param: 'tags'` → options derived from curationStore.get(hash).tags
- `param: 'resolution'` → resolutionPairs has one pair per unique (width, height)
- `param: 'cfg'` (numeric) → options is empty (numeric params don't use this path)

import { computed } from 'vue'
import { useI18n } from 'vue-i18n'
import MoshpitAddFilterPopover from './MoshpitAddFilterPopover.vue'
import { useMoshpitFilterStore } from '@/platform/moshpit/stores/moshpitFilterStore'

const filterStore = useMoshpitFilterStore()

function chipLabel(chip: FilterChip): string {
  // Map chip.param to the i18n display name per UI-SPEC Surface 5 mapping
  const key = `moshpit.filters.param${chip.param.charAt(0).toUpperCase()}${chip.param.slice(1)}`
  // Known keys: paramModel/paramLoras/paramCfg/paramSteps/paramSampler/paramScheduler/paramSeed/paramPrompt/paramNegativePrompt/paramResolution/paramGenerationTime/paramTags/paramFavourite
  // Special mapping for positivePrompt → paramPrompt; negativePrompt → paramNegativePrompt; timestamp → paramGenerationTime
  switch (chip.param) {
    case 'positivePrompt': return t('moshpit.filters.paramPrompt')
    case 'negativePrompt': return t('moshpit.filters.paramNegativePrompt')
    case 'timestamp': return t('moshpit.filters.paramGenerationTime')
    case 'width':
    case 'height':
    case 'resolution': return t('moshpit.filters.paramResolution')
    default: return t(key)
  }
}

function chipValueSummary(chip: FilterChip): string {
  // Per UI-SPEC Surface 4 "Chip value summary format":
  switch (chip.value.kind) {
    case 'categorical': {
      if (chip.value.values.length === 0) return ''
      if (chip.value.values.length === 1) return chip.value.values[0]
      return `${chip.value.values[0]} +${chip.value.values.length - 1}`
    }
    case 'numeric': {
      if (chip.value.exact !== null) return `= ${chip.value.exact}`
      const min = chip.value.min === null ? '' : String(chip.value.min)
      const max = chip.value.max === null ? '' : String(chip.value.max)
      return `${min}–${max}`
    }
    case 'text':
      return `"${chip.value.substring}"`
    case 'resolution': {
      if (chip.value.pairs.length === 0) return ''
      const first = `${chip.value.pairs[0][0]}×${chip.value.pairs[0][1]}`
      return chip.value.pairs.length === 1 ? first : `${first} +${chip.value.pairs.length - 1}`
    }
    case 'boolean':
      return chip.value.value ? t('moshpit.filters.paramFavourite') : ''
  }
}
</script>

function paramLabel(p: ParamKey): string {
  const entry = PARAM_ENTRIES.find((e) => e.key === p)
  return entry ? t(`moshpit.filters.${entry.label}`) : p
}

const activeParams = computed(() => new Set(filterStore.chips.map((c) => c.param)))

const filteredParams = computed(() => {
  const base = PARAM_ENTRIES.map((p) => ({
    ...p,
    label: t(`moshpit.filters.${p.label}`),
    alreadyActive: activeParams.value.has(p.key)
  }))
  const q = paramSearch.value.trim()
  if (!q) return base
  const fuse = new Fuse(base, { keys: ['label'], threshold: 0.4 })
  return fuse.search(q).map((r) => r.item)
})

const editorKind = computed<'numeric' | 'categorical' | 'text' | 'resolution' | 'boolean' | null>(() => {
  if (!draftParam.value) return null
  const entry = PARAM_ENTRIES.find((e) => e.key === draftParam.value)
  if (!entry) return null
  return entry.kind === 'time' ? null : entry.kind
})

// Value-editor state — reset when draftParam changes
const numericMin = ref<number | null>(null)
const numericMax = ref<number | null>(null)
const numericExact = ref<number | null>(null)
const useExact = ref(false)
const textSubstring = ref('')
const selectedValues = ref<string[]>([])
const selectedResolutions = ref<string[]>([])
const customW = ref<number | null>(null)
const customH = ref<number | null>(null)
const booleanValue = ref(true)
const valueSearch = ref('')

const { options: valueOptions, resolutionPairs } = useMoshpitParamValueOptions(computed(() => draftParam.value).value as ParamKey | null)
// NOTE: useMoshpitParamValueOptions accepts ParamKey | null — pass the current draftParam directly; re-instantiation triggers via watch below.

const filteredValueOptions = computed(() => {
  const q = valueSearch.value.trim()
  if (!q) return valueOptions.value
  const fuse = new Fuse([...valueOptions.value], { keys: ['value'], threshold: 0.4 })
  return fuse.search(q).map((r) => r.item)
})

const canApply = computed(() => {
  switch (editorKind.value) {
    case 'numeric':
      return useExact.value
        ? typeof numericExact.value === 'number' && !Number.isNaN(numericExact.value)
        : (numericMin.value !== null || numericMax.value !== null)
    case 'text': return textSubstring.value.trim().length > 0
    case 'categorical': return selectedValues.value.length > 0
    case 'resolution':
      return selectedResolutions.value.length > 0 || (customW.value !== null && customH.value !== null)
    case 'boolean': return true
    default: return false
  }
})

function resetValueState(): void {
  numericMin.value = null; numericMax.value = null; numericExact.value = null
  useExact.value = false; textSubstring.value = ''; selectedValues.value = []
  selectedResolutions.value = []; customW.value = null; customH.value = null
  booleanValue.value = true; valueSearch.value = ''
}

watch(draftParam, () => resetValueState())

function selectParam(key: ParamKey): void {
  if (activeParams.value.has(key)) return
  draftParam.value = key
  step.value = 'value'
}

function backToStep1(): void {
  step.value = 'param'
  draftParam.value = null
}

function buildChipValue(): ChipValue | null {
  switch (editorKind.value) {
    case 'numeric':
      if (useExact.value) return { kind: 'numeric', min: null, max: null, exact: numericExact.value }
      return { kind: 'numeric', min: numericMin.value, max: numericMax.value, exact: null }
    case 'categorical':
      return { kind: 'categorical', values: [...selectedValues.value] }
    case 'text':
      return { kind: 'text', substring: textSubstring.value.trim() }
    case 'resolution': {
      const pairs: [number, number][] = []
      for (const key of selectedResolutions.value) {
        const [w, h] = key.split('x').map(Number)
        if (Number.isFinite(w) && Number.isFinite(h)) pairs.push([w, h])
      }
      if (customW.value !== null && customH.value !== null) pairs.push([customW.value, customH.value])
      return { kind: 'resolution', pairs }
    }
    case 'boolean':
      return { kind: 'boolean', value: booleanValue.value }
    default:
      return null
  }
}

function onApply(): void {
  if (!draftParam.value || !canApply.value) return
  const value = buildChipValue()
  if (!value) return
  const chip: FilterChip = {
    id: typeof crypto !== 'undefined' && 'randomUUID' in crypto ? crypto.randomUUID() : `chip-${Date.now()}-${Math.random()}`,
    param: draftParam.value,
    value
  }
  filterStore.addChip(chip)
  isOpen.value = false
  backToStep1()
}
