**E. Workflow-selection → queue.setFilter wiring** in MoshpitLayout:

```ts
const filterStore = useMoshpitFilterStore()
const assetsStore = useAssetsStore()
const metaStore = useMoshpitMetadataStore()

watch(
  () => [filterStore.workflow, filterStore.timeRange] as const,
  ([workflow, timeRange]) => {
    if (!workflow) return
    const nowMs = Date.now()
    const range = getDateRangeForPreset(timeRange.preset, nowMs)
    const fromMs = timeRange.preset === 'custom' ? (timeRange.from ?? -Infinity) : (range?.from ?? -Infinity)
    const toMs = timeRange.preset === 'custom' ? (timeRange.to ?? Infinity) : (range?.to ?? Infinity)

    const candidates = assetsStore.outputJobAssets.filter((a) => {
      const created = a.created_at ? new Date(a.created_at).getTime() : 0
      if (!(created >= fromMs && created <= toMs)) return false
      const hash = a.asset_hash ?? metaStore.getHashForAssetId(a.id)
      if (!hash) return true   // un-processed assets pass through to be filtered after thumbReady
      const params = metaStore.paramsByHash.get(hash)
      if (!params) return true
      return params.workflowFingerprint === workflow
    })
    const filterKey = `${workflow}:${timeRange.preset}:${timeRange.from ?? ''}-${timeRange.to ?? ''}`
    void queue.setFilter(filterKey, candidates)
  },
  { immediate: false }
)
```

Note: `getDateRangeForPreset` is imported from `@/platform/moshpit/services/filterMath`.

**Important:** Un-processed assets (no params yet) cannot be filtered by fingerprint. They get enqueued optimistically — once their params land via thumbReady, the in-memory filter (via useMoshpitFilteredAssets) excludes them from the visible set if their fingerprint mismatches. This respects D-06 "best-effort" thinking and avoids a chicken-and-egg where the gate needs params to know what to process.
