    Implementation — each "many" verb follows this template:
    ```ts
    function favouriteMany(hashes: readonly string[], favourite = true): void {
      if (hashes.length === 0) return
      // 1. Capture inverse
      const prior = new Map<string, CurationRecord>()
      const next = new Map<string, CurationRecord>()
      for (const h of hashes) {
        const cur = curationStore.get(h) ?? defaultCuration()
        prior.set(h, cur)
        if (cur.favourite !== favourite) {
          next.set(h, { ...cur, favourite })
        }
      }
      if (next.size === 0) return  // no-op: all already at target

      // 2. Apply
      void curationStore.applyManyOptimistic(next)

      // 3. Toast ONLY if bulk (hashes.length > 1) — CURATE-06
      if (hashes.length > 1) {
        publishUndo({
          label: t(favourite ? 'moshpit.curation.favourited' : 'moshpit.curation.unfavourited', { count: next.size }),
          undo: () => {
            const restore = new Map<string, CurationRecord>()
            for (const [h, p] of prior) restore.set(h, p)
            void curationStore.applyManyOptimistic(restore)
          }
        })
      }
    }
    ```

      const resolver = options.resolveFullResUrl
      if (!resolver) {

    publishUndo helper:
    ```ts
    function publishUndo(params: { label: string; undo: () => void | Promise<void> }): void {
      const action: UndoableAction = {
        id: crypto.randomUUID(),
        label: params.label,
        expiresAt: Date.now() + UNDO_WINDOW_MS,
        undo: params.undo
      }
      lastUndoable.value = action
      toastStore.add({
        severity: 'info',
        summary: action.label,
        life: UNDO_WINDOW_MS,
        group: 'moshpit-curation'
      })
    }
    ```

    undoLast:
    ```ts
    function undoLast(): boolean {
      const a = lastUndoable.value
      if (!a) return false
      if (Date.now() > a.expiresAt) {
        lastUndoable.value = null
        return false
      }
      void a.undo()
      lastUndoable.value = null
      // close the pending toast — use removeGroup if available, otherwise removeAll.
      // Verify toastStore API at implementation time; fall back to removeAll.
      if (typeof toastStore.removeGroup === 'function') toastStore.removeGroup('moshpit-curation')
      else toastStore.removeAll?.()
      return true
    }
    ```
