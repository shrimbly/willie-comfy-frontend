    ```ts
    import { useMoshpitCuration } from '@/platform/moshpit/composables/useMoshpitCuration'
    import { useMoshpitCurationStore } from '@/platform/moshpit/stores/moshpitCurationStore'

    // existing actions/selection/overrideStore already wired
    const curation = useMoshpitCuration()
    const curationStore = useMoshpitCurationStore()

    const emit = defineEmits<{
      'open-tag-popover': [payload: { hashes: readonly string[]; anchorX: number; anchorY: number }]
      'open-folder-picker': [payload: { hashes: readonly string[]; anchorX: number; anchorY: number }]
    }>()

    const allFavourited = computed(() =>
      actionSet.value.length > 0 &&
      actionSet.value.every((h) => curationStore.get(h)?.favourite === true)
    )
    const allHidden = computed(() =>
      actionSet.value.length > 0 &&
      actionSet.value.every((h) => curationStore.get(h)?.hidden === true)
    )

    const favouriteLabel = computed(() =>
      allFavourited.value ? t('moshpit.contextMenu.unfavourite') : t('moshpit.contextMenu.favourite')
    )
    const hideLabel = computed(() =>
      allHidden.value ? t('moshpit.contextMenu.unhide') : t('moshpit.contextMenu.hide')
    )

    function onFavourite(): void {
      curation.favouriteMany(actionSet.value, !allFavourited.value)
      close()
    }
    function onTag(): void {
      emit('open-tag-popover', {
        hashes: [...actionSet.value],
        anchorX: anchorX.value,
        anchorY: anchorY.value
      })
      close()
    }
    function onHide(): void {
      curation.hideMany(actionSet.value, !allHidden.value)   // Plan 03: hideMany(hashes, hidden=true)
      close()
    }
    function onFolder(): void {
      emit('open-folder-picker', {
        hashes: [...actionSet.value],
        anchorX: anchorX.value,
        anchorY: anchorY.value
      })
      close()
    }
    function onExport(): void {
      curation.exportMany(actionSet.value)   // resolveFullResUrl must be supplied — MoshpitView passes via the shared composable call. If not available, useMoshpitCuration logs a warn.
      close()
    }
    ```

    ```ts
    import { useMoshpitFoldersStore } from '@/platform/moshpit/stores/moshpitFoldersStore'
    import { useToastStore } from '@/platform/updates/common/toastStore'

    const foldersStore = useMoshpitFoldersStore()
    const toastStore = useToastStore()

    const savingFolder = ref(false)
    const folderName = ref('')
    watch(savingFolder, (val) => {
      if (val) {
        // seed default "Tournament Apr 23, 14:03"
        const now = new Date()
        folderName.value = t('moshpit.tournament.winner.saveAsFolderDefault', {
          date: now.toLocaleString()
        })
      }
    })

    function onSaveFolder(): void {
      const trimmed = folderName.value.trim()
      if (!trimmed || trimmed.length > 64) {
        toastStore.add({
          severity: 'warn',
          summary: t('moshpit.curation.invalidFolder'),
          life: 3000
        })
        return
      }
      const winners = [...tournamentStore.winnerHashes]
      foldersStore.createFromSelection(trimmed, winners)
      toastStore.add({
        severity: 'info',
        summary: t('moshpit.curation.folderCreated', {
          name: trimmed,
          count: winners.length
        }),
        life: 4000
      })
      savingFolder.value = false
      folderName.value = ''
    }
    ```
