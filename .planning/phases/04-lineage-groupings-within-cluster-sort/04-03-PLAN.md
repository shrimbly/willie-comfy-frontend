```typescript
      return rec ? [[h, rec] as const] : []
    })
  )

  const filtered = applyFilterChips(
    hashToParams,
    hashToCuration,
    filterStore.chips,
    filterStore.showHidden,
    filterStore.timeRange,
    Date.now()
  )

  const registrySet = new Set(visibleHashes)
  const visible = filtered.filter((h) => registrySet.has(h))

  // Build filename map (null-safe); upstream shape varies — if registry entry
  // has a filename field, use it, else fall back to null.
  const filenameByHash = new Map<string, string | null>()
  for (const e of allEntries) {
    // NOTE: verify the actual field name in useMoshpitAssetRegistry before
    // committing — may be e.filename / e.sourceFilename / not present at all.
    // If not present, leave this loop empty and every entry maps to null.
    filenameByHash.set(e.contentHash, (e as { filename?: string | null }).filename ?? null)
  }

  const nestingOrder = computeNestingOrder(
    visible,
    hashToParams,
    filenameByHash,
    filterStore.activeGroupings
  )

  const { root, slots } = computeClusterLayout(
    visible,
    hashToParams,
    filenameByHash,
    nestingOrder,
    filterStore.withinClusterSort,
    filterStore.gridSpacing
  )

  return {
    visible: slots.map((s) => s.hash),
    slotByHash: new Map(slots.map((s) => [s.hash, s])),
    clusterTree: root
  }
})

const entries = computed<readonly FilteredAssetEntry[]>(() => {
  const src = registry.entries.value
  const { slotByHash } = layout.value
  const out: FilteredAssetEntry[] = []
  for (const e of src) {
    const slot = slotByHash.get(e.contentHash)
    if (!slot) continue
    out.push({
      id: e.id,
      contentHash: e.contentHash,
      thumbUrl: e.thumbUrl,
      worldX: slot.worldX,
      worldY: slot.worldY
    })
  }
  return out
})

const clusterTree = computed<ClusterNode | null>(() => layout.value.clusterTree)

const activeGroupingOrder = computed<readonly GroupingAxis[]>(() => {
  if (!filterStore.isGated) return []
  const allEntries = registry.entries.value
  const hashToParams = metaStore.paramsByHash
  const visibleHashes = allEntries.map((e) => e.contentHash)
  const filenameByHash = new Map<string, string | null>(
    allEntries.map((e) => [e.contentHash, (e as { filename?: string | null }).filename ?? null])
  )
  return computeNestingOrder(visibleHashes, hashToParams, filenameByHash, filterStore.activeGroupings)
})

return { entries, clusterTree, activeGroupingOrder }
```

Note: D-05 / Pitfall 6 warn about double-computing bucket keys. The simplest correct implementation is the above — `computeClusterLayout` already memoises internally (Plan 01 Task 2). The `activeGroupingOrder` computed re-walks the list; this is acceptable because the Plan 04 UI only reads `activeGroupingOrder` for debug / accessibility labels (not rendering), and Vue memoises the computed between reactive ticks.
