<objective>
Ship the two visual primitives that render a tournament pair:

1. `MoshpitTournamentAssetFrame.vue` — a single-asset DOM frame that starts with a thumbnail (cached WebP from `moshpitThumbStore`) and crossfades to the full-res asset URL when the preloaded `Image()` finishes (D-24). Reused twice by the Pair component.

Purpose: decouples pixel-rendering concerns from overlay chrome (legend, pair counter, peek panel). Pair stays stateless and reactive to store.

Output: two Vue 3.5 SFCs, two test files, driven by store state from Plan 05-03.
</objective>

       ```vue
       <script setup lang="ts">
       import { computed, ref, watch } from 'vue'
       import { useI18n } from 'vue-i18n'
       import { cn } from '@/utils/tailwindUtil'
       import { useMoshpitThumbStore } from '@/platform/moshpit/stores/moshpitThumbStore'

       const {
         hash,
         fullResUrl,
         label,
         highlight = false
       } = defineProps<{
         hash: string
         fullResUrl: string | null
         label?: 'A' | 'B'
         highlight?: boolean
       }>()

       const { t } = useI18n()
       const thumbStore = useMoshpitThumbStore()

       const thumbUrl = computed<string | undefined>(() => thumbStore.getUrl(hash))
       const isFullResLoaded = ref(false)

       // Preload full-res into a detached Image; swap `src` on load.
       watch(
         () => fullResUrl,
         (url) => {
           isFullResLoaded.value = false
           if (!url) return
           const preloader = new Image()
           preloader.onload = () => { isFullResLoaded.value = true }
           preloader.onerror = () => { /* stay on thumb */ }
           preloader.src = url
         },
         { immediate: true }
       )

       const shownSrc = computed<string | undefined>(() =>
         isFullResLoaded.value && fullResUrl ? fullResUrl : thumbUrl.value
       )

       const altText = computed(() => t('moshpit.tournament.assetAlt'))
       </script>

       <template>
         <div
           :class="cn(
             'relative size-full flex items-center justify-center bg-background transition-all duration-300 ease-out',
             highlight && 'ring-2 ring-primary'
           )"
         >
           <img
             v-if="shownSrc"
             :src="shownSrc"
             :alt="altText"
             :class="cn(
               'max-w-full max-h-full size-full object-contain transition-opacity duration-300 ease-out',
               isFullResLoaded ? 'opacity-100' : 'opacity-95'
             )"
             draggable="false"
           />
           <div
             v-if="label"
             class="absolute top-2 left-2 px-2 py-1 text-xs font-medium bg-background/80 rounded"
           >
             {{ label }}
           </div>
         </div>
       </template>
       ```

    ```vue
    <script setup lang="ts">
    import { computed, ref } from 'vue'
    import { cn } from '@/utils/tailwindUtil'
    import MoshpitTournamentAssetFrame from './MoshpitTournamentAssetFrame.vue'
    import { useMoshpitTournamentStore } from '@/platform/moshpit/stores/moshpitTournamentStore'

    const { resolveFullResUrl } = defineProps<{
      resolveFullResUrl: (hash: string) => string | null
    }>()

    const store = useMoshpitTournamentStore()

    const rootEl = ref<HTMLElement | null>(null)
    const isDraggingDivider = ref(false)

    function urlFor(hash: string): string | null {
      return resolveFullResUrl(hash)
    }

    function onDividerPointerDown(e: PointerEvent) {
      isDraggingDivider.value = true
      ;(e.currentTarget as Element).setPointerCapture?.(e.pointerId)
    }
    function onDividerPointerMove(e: PointerEvent) {
      if (!isDraggingDivider.value || !rootEl.value) return
      const rect = rootEl.value.getBoundingClientRect()
      const fraction = (e.clientX - rect.left) / rect.width
      store.setWipePosition(fraction)
    }
    function onDividerPointerUp() {
      isDraggingDivider.value = false
    }

    const clipStyleForB = computed(() => ({
      clipPath: `inset(0 ${store.wipePosition * 100}% 0 0)`
    }))
    </script>

    <template>
      <div ref="rootEl" class="relative size-full flex items-stretch bg-background">
        <template v-if="store.currentPair">
          <!-- sideBySide -->
          <template v-if="store.displayMode === 'sideBySide'">
            <div class="w-1/2 h-full">
              <MoshpitTournamentAssetFrame
                :hash="store.currentPair.assetHashA"
                :fullResUrl="urlFor(store.currentPair.assetHashA)"
                label="A"
                :highlight="!store.flipShowsB"
              />
            </div>
            <div class="w-1/2 h-full">
              <MoshpitTournamentAssetFrame
                :hash="store.currentPair.assetHashB"
                :fullResUrl="urlFor(store.currentPair.assetHashB)"
                label="B"
                :highlight="store.flipShowsB"
              />
            </div>
          </template>

          <!-- overlap -->
          <template v-else-if="store.displayMode === 'overlap'">
            <div class="absolute inset-0">
              <MoshpitTournamentAssetFrame
                :hash="store.currentPair.assetHashA"
                :fullResUrl="urlFor(store.currentPair.assetHashA)"
                label="A"
              />
            </div>
            <div class="absolute inset-0" :style="clipStyleForB">
              <MoshpitTournamentAssetFrame
                :hash="store.currentPair.assetHashB"
                :fullResUrl="urlFor(store.currentPair.assetHashB)"
                label="B"
              />
            </div>
            <div
              class="absolute top-0 bottom-0 w-0.5 bg-foreground pointer-events-auto cursor-ew-resize"
              :style="{ left: `${store.wipePosition * 100}%` }"
              @pointerdown="onDividerPointerDown"
              @pointermove="onDividerPointerMove"
              @pointerup="onDividerPointerUp"
              @pointercancel="onDividerPointerUp"
            >
              <div class="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 size-6 rounded-full bg-background border border-foreground" />
            </div>
          </template>

          <!-- flip -->
          <template v-else-if="store.displayMode === 'flip'">
            <MoshpitTournamentAssetFrame
              :hash="store.flipShowsB ? store.currentPair.assetHashB : store.currentPair.assetHashA"
              :fullResUrl="urlFor(store.flipShowsB ? store.currentPair.assetHashB : store.currentPair.assetHashA)"
              :label="store.flipShowsB ? 'B' : 'A'"
            />
          </template>
        </template>
      </div>
    </template>
    ```
