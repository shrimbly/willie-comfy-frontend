<objective>
Ship the pure bracket math that powers tournament mode. Two responsibilities:
(1) turn an ordered array of asset hashes into a deterministic sequence of
pairwise matches (round-robin for N<8, single-elim for N>=8), and
(2) fold pick / skip events into a wins map and derive the final winner set.

Purpose: isolates the deterministic, replayable game logic from Vue / Pinia
so it is unit-testable with `fast-check` property tests and stays worker-
safe for a potential future v2-TOUR-01 offline elo implementation.

Output: `tournamentBracket.ts` + `tournamentBracket.test.ts` covering D-01
through D-09. No UI, no store — those arrive in Plan 03.

```typescript
// Contracts this plan MUST export from tournamentBracket.ts

export type BracketShape = 'roundRobin' | 'singleElim'

export interface TournamentPair {
  readonly seedA: number
  readonly seedB: number
  readonly assetHashA: string
  readonly assetHashB: string
  readonly index: number // 0-based position in the current queue
  readonly total: number // total pairs for round-robin; total picks expected for single-elim
}

export const ROUND_ROBIN_THRESHOLD = 8 // D-01 tunable constant

export function decideBracketShape(n: number): BracketShape

// Round-robin: lexicographic (0,1),(0,2),...,(0,N-1),(1,2),...
// Single-elim: N-1 matches; byes to first (nextPow2(N)-N) seeds get walkover wins in round 1.
//   The returned array is the INITIAL queue — downstream rounds are produced by applyPick.
export function generateInitialBracket(
  hashes: readonly string[],
  shape: BracketShape
): readonly TournamentPair[]

// Result of applying a pick to the wins map (immutable update).
// LOCKED CONTRACT: queueTail is ALWAYS empty — next-round generation lives in
// the separate pure helper `generateNextRound`. The store (Plan 03) tracks
// current-round winners and calls generateNextRound at the round boundary.
export interface BracketUpdate {
  readonly wins: ReadonlyMap<string, number>
  readonly queueTail: readonly TournamentPair[] // ALWAYS empty in this contract; kept in the shape for future-compat
}

export function applyPick(
  wins: ReadonlyMap<string, number>,
  pair: TournamentPair,
  winner: 'A' | 'B',
  shape: BracketShape,
  remainingQueueAfterThis: readonly TournamentPair[]
): BracketUpdate

// Generates the next single-elim round given the current round's winners in
// pick order. Pure: zero dependency on `wins` or the live queue. Pair indices
// start at `baseIndex` so they stay unique across the bracket timeline.
export function generateNextRound(
  currentRoundWinnersInOrder: readonly string[],
  totalPicksExpected: number,
  baseIndex: number
): readonly TournamentPair[]

// For round-robin: move pair to tail; returns [newQueue, sameWins, sameSkipIndexes + thisIndex].
// For single-elim: bye for seed A; pair.index recorded in skippedPairIndexes.
export interface SkipUpdate {
  readonly nextQueue: readonly TournamentPair[]
  readonly skippedPairIndexes: ReadonlySet<number>
  readonly wins: ReadonlyMap<string, number> // single-elim may record bye-win for A
}

export function applySkip(
  pair: TournamentPair,
  remainingQueueAfterThis: readonly TournamentPair[],
  skippedPairIndexes: ReadonlySet<number>,
  wins: ReadonlyMap<string, number>,
  shape: BracketShape
): SkipUpdate

export function computeWinnerSet(
  wins: ReadonlyMap<string, number>,
  shape: BracketShape,
  hashesInEntryOrder: readonly string[],
  topN?: number // defaults to 3 for round-robin; ignored for single-elim
): readonly string[]
```
