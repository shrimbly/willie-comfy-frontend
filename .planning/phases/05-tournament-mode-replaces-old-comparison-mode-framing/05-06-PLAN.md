```typescript
// New handler on the container element
function onContainerKeydown(e: KeyboardEvent) {
  if (e.key !== 'Enter') return
  if (tournamentStore.isActive) return
  if (selectionStore.size < 2) {
    toastStore.add({
      severity: 'info',
      summary: t('moshpit.tournament.needTwoToastSummary'),
      detail: t('moshpit.tournament.needTwoToastDetail')
    })
    e.preventDefault()
    return
  }
  tournamentStore.enter(selectionStore.selected, resolveFullResUrlFactory())
  e.preventDefault()
}
```
