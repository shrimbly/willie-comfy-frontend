    ```vue
    <script setup lang="ts">
    import { computed } from 'vue'
    import { useI18n } from 'vue-i18n'
    import { cn } from '@/utils/tailwindUtil'
    import {
      diffParams,
      diffLoras,
      PARAM_DIFF_KEY_ORDER,
      type LoraDiffEntry,
      type ParamDiffRow,
      type ParamDiffState,
      type LoraDiffState
    } from '@/platform/moshpit/services/metadataDiff'
    import type { NormalizedParams } from '@/platform/moshpit/services/paramNormalize'
    import { useMoshpitTournamentStore } from '@/platform/moshpit/stores/moshpitTournamentStore'

    const { paramsA, paramsB } = defineProps<{
      paramsA: NormalizedParams | null
      paramsB: NormalizedParams | null
    }>()

    const { t } = useI18n()
    const store = useMoshpitTournamentStore()

    const isReady = computed(() => paramsA !== null && paramsB !== null)

    const paramRows = computed<readonly ParamDiffRow[]>(() =>
      paramsA && paramsB ? diffParams(paramsA, paramsB) : []
    )

    const loraRows = computed<readonly LoraDiffEntry[]>(() =>
      paramsA && paramsB ? diffLoras(paramsA.loras, paramsB.loras) : []
    )

    function paramStateClass(state: ParamDiffState): string {
      return state === 'differ' || state === 'onlyA' || state === 'onlyB'
        ? 'bg-node-component-surface'
        : ''
    }

    function loraStateClass(state: LoraDiffState): string {
      switch (state) {
        case 'added': return 'text-success'
        case 'removed': return 'text-danger'
        case 'weightChanged': return 'text-warning'
        default: return 'text-muted-foreground'
      }
    }

    function formatValue(v: unknown): string {
      if (v === null || v === undefined || v === '') return '—'
      if (typeof v === 'number') return String(v)
      if (typeof v === 'string') return v
      return JSON.stringify(v)
    }
    </script>

    <template>
      <aside
        :class="cn(
          'absolute top-0 right-0 h-full w-80 lg:w-96 xl:w-[28rem] bg-background border-l border-border overflow-y-auto transition-[transform] duration-200 ease-out',
          store.isPeekOpen ? 'translate-x-0' : 'translate-x-full'
        )"
        :aria-hidden="!store.isPeekOpen"
      >
        <header class="p-4 border-b border-border">
          <h2 class="text-sm font-medium">{{ t('moshpit.peek.panelTitle') }}</h2>
        </header>

        <section v-if="!isReady" class="p-4 text-sm text-muted-foreground">
          {{ t('moshpit.peek.loadingMetadata') }}
        </section>

        <section v-else class="p-4 flex flex-col gap-1">
          <div
            v-for="row in paramRows"
            :key="row.key"
            :class="cn('grid grid-cols-[1fr_auto_1fr] gap-2 px-2 py-1 rounded text-xs', paramStateClass(row.state))"
          >
            <span class="text-muted-foreground">{{ t(`moshpit.peek.params.${row.key}`) }}</span>
            <span class="font-mono">{{ formatValue(row.valueA) }}</span>
            <span class="font-mono">{{ formatValue(row.valueB) }}</span>
          </div>
        </section>

        <section v-if="isReady" class="p-4 border-t border-border">
          <h3 class="text-xs font-medium mb-2">{{ t('moshpit.peek.lora.sectionTitle') }}</h3>
          <div v-if="loraRows.length === 0" class="text-xs text-muted-foreground">
            {{ t('moshpit.peek.lora.empty') }}
          </div>
          <ul v-else class="flex flex-col gap-1 text-xs">
            <li v-for="row in loraRows" :key="row.name" :class="cn('px-2 py-1 rounded', loraStateClass(row.state))">
              <span class="font-mono">{{ row.name }}</span>
              <span class="ml-2">
                {{ formatValue(row.weightA) }} → {{ formatValue(row.weightB) }}
              </span>
              <span class="ml-2 text-muted-foreground">
                {{ t(`moshpit.peek.lora.state${row.state.charAt(0).toUpperCase() + row.state.slice(1)}`) }}
              </span>
            </li>
          </ul>
        </section>
      </aside>
    </template>
    ```
