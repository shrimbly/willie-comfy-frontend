```typescript
// Contracts this plan MUST export from metadataDiff.ts

import type { NormalizedParams } from './paramNormalize'

export type ParamDiffState =
  | 'match' // both have same value
  | 'differ' // both have values that differ
  | 'onlyA' // only A has a value
  | 'onlyB' // only B has a value
  | 'missingBoth' // neither has a value

export type ParamDiffKey = Exclude<keyof NormalizedParams, 'loras'>

export interface ParamDiffRow {
  readonly key: ParamDiffKey
  readonly state: ParamDiffState
  readonly valueA: unknown
  readonly valueB: unknown
}

// Canonical render order for the peek panel. First entry is shown first.
// Matches the conceptual ordering in paramNormalize.ts (model, loras handled separately, then sampler/scheduler, then numerics, then prompts, then identity).
export const PARAM_DIFF_KEY_ORDER: readonly ParamDiffKey[]

export function diffParams(
  a: NormalizedParams,
  b: NormalizedParams
): readonly ParamDiffRow[]

export type LoraDiffState = 'match' | 'weightChanged' | 'added' | 'removed'

export interface LoraDiffEntry {
  readonly name: string
  readonly state: LoraDiffState
  readonly weightA: number | null // null when not present in A
  readonly weightB: number | null // null when not present in B
}

export function diffLoras(
  a: readonly { readonly name: string; readonly weight: number }[],
  b: readonly { readonly name: string; readonly weight: number }[]
): readonly LoraDiffEntry[]
```

    ```typescript
    import { sortBy } from 'es-toolkit'
    import type { NormalizedParams } from './paramNormalize'

    export type ParamDiffKey = Exclude<keyof NormalizedParams, 'loras'>

    // Render order — matches paramNormalize.ts field order minus 'loras' (which is rendered via diffLoras).
    // KEEP ALIGNED with NormalizedParamsSchema field order.
    export const PARAM_DIFF_KEY_ORDER: readonly ParamDiffKey[] = [
      'model',
      'cfg',
      'steps',
      'sampler',
      'scheduler',
      'seed',
      'width',
      'height',
      'positivePrompt',
      'negativePrompt',
      'timestamp',
      'workflowFingerprint',
      'workflowFilename',
      'saveNodeIdentity'
    ]

    export type ParamDiffState = 'match' | 'differ' | 'onlyA' | 'onlyB' | 'missingBoth'

    export interface ParamDiffRow {
      readonly key: ParamDiffKey
      readonly state: ParamDiffState
      readonly valueA: unknown
      readonly valueB: unknown
    }

    export function diffParams(a: NormalizedParams, b: NormalizedParams): readonly ParamDiffRow[] {
      return PARAM_DIFF_KEY_ORDER.map((key) => {
        const va = a[key]
        const vb = b[key]
        const aMissing = va === undefined || va === null || va === ''
        const bMissing = vb === undefined || vb === null || vb === ''
        let state: ParamDiffState
        if (aMissing && bMissing) state = 'missingBoth'
        else if (aMissing) state = 'onlyB'
        else if (bMissing) state = 'onlyA'
        else state = va === vb ? 'match' : 'differ'
        return { key, state, valueA: va, valueB: vb }
      })
    }
    ```
