      ```typescript
      function nextPow2(n: number): number {
        return n <= 1 ? 1 : 1 << Math.ceil(Math.log2(n))
      }

      function singleElimRoundSizes(entryCount: number): number[] {
        if (entryCount < 2) return []
        const sizes: number[] = []
        const byeCount = nextPow2(entryCount) - entryCount
        const firstRound = (entryCount - byeCount) / 2
        if (firstRound > 0) sizes.push(firstRound)
        let size = nextPow2(entryCount) / 4
        while (size >= 1) {
          sizes.push(size)
          size = size / 2
        }
        return sizes
      }

      function sliceIntoRounds(
        pairs: readonly TournamentPair[],
        sizes: readonly number[]
      ): TournamentPair[][] {
        const rounds: TournamentPair[][] = []
        let cursor = 0
        for (const size of sizes) {
          if (cursor >= pairs.length) break
          const end = Math.min(pairs.length, cursor + size)
          rounds.push([...pairs.slice(cursor, end)])
          cursor = end
          if (cursor - (end - size) < size) break // partial round: stop here
        }
        return rounds
      }
      ```
      Note: the partial-round guard above stops at the first incomplete round so we never render a half-filled round that hasn't had picks yet. Simpler equivalent: break out of the loop when `cursor < (previous cursor + size)` i.e. `pairs.length < cursor + size`. Use the simpler form:
      ```typescript
      for (const size of sizes) {
        if (cursor + size > pairs.length) break
        rounds.push([...pairs.slice(cursor, cursor + size)])
        cursor += size
      }
      ```
      (This means a round only shows up once ALL its pairs have been generated — matches how the store appends a full next round at the boundary.)
